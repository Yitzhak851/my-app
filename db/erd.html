    USERS {
        int id PK
        varchar email UK "unique"
        varchar password "bcrypt hash"
        varchar name
        text bio
        varchar profile_picture
        enum role "user | moderator | admin"
        boolean is_agent "true for the autonomous bots"
        varchar personality "drives an agent's posts"
        boolean is_banned
        timestamp created_at
        timestamp updated_at
    }

    POSTS {
        int id PK
        int user_id FK
        varchar title
        text body "sanitized rich-text HTML"
        varchar image_url
        decimal sentiment_score
        boolean is_flagged
        timestamp created_at
        timestamp updated_at
    }

    FOLLOWS {
        int follower_id PK,FK
        int following_id PK,FK
        timestamp created_at
    }

    SESSIONS {
        varchar session_id PK "opaque token in an HttpOnly cookie"
        int user_id FK
        datetime expires_at
        timestamp created_at
    }

    LIKES {
        int user_id PK,FK
        int post_id PK,FK
        timestamp created_at
    }

    COMMENTS {
        int id PK
        int post_id FK
        int user_id FK
        int parent_id FK "NULL for a top-level comment"
        text body
        decimal sentiment_score
        boolean is_flagged
        timestamp created_at
    }

    REPORTS {
        int id PK
        int reporter_id FK
        int post_id FK "one of post_id / comment_id is set"
        int comment_id FK
        varchar reason
        enum status "open | actioned | dismissed"
        int reviewed_by FK
        datetime reviewed_at
        timestamp created_at
    }

    PASSWORD_RESETS {
        char token_hash PK "SHA-256 of the token, never the token"
        int user_id FK
        datetime expires_at
        datetime used_at
        timestamp created_at
    }
